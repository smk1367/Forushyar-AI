def convert_wav_to_8k(src: bytes, target: Path) -> None:
    """Band-limited PCM16 WAV -> mono PCM16 WAV at 8000 Hz.

    Windowed-sinc FIR anti-aliasing is essential for 22050/24000 Hz Piper speech.
    Self-contained: works in the existing pilot container without new packages.
    """
    import io
    import math

    with wave.open(io.BytesIO(src), "rb") as inp:
        nch, width, rate = inp.getnchannels(), inp.getsampwidth(), inp.getframerate()
        if nch not in (1, 2) or width != 2 or rate < 8000 or inp.getcomptype() != "NONE":
            raise ValueError("Unsupported TTS WAV format")
        frames = inp.getnframes()
        if frames > rate * 35 or frames < 2:
            raise ValueError("Invalid TTS audio duration")
        pcm = array("h")
        pcm.frombytes(inp.readframes(frames))

    if sys.byteorder != "little":
        pcm.byteswap()
    n = len(pcm) // nch
    if n < 2:
        raise ValueError("Empty TTS audio")
    # Perform stereo->mono once, not in every interpolation tap.
    if nch == 2:
        samples = [(int(pcm[i]) + int(pcm[i + 1])) * 0.5 for i in range(0, n * 2, 2)]
    else:
        samples = pcm

    output = array("h")
    if rate == 8000:
        output.extend(max(-32768, min(32767, round(x))) for x in samples)
    else:
        half = 24
        phases = 256
        cutoff = 0.45 * 8000.0 / rate
        taps = range(-half + 1, half + 1)
        # Precomputed polyphase filters: prevents repeated trig during a call.
        table = []
        for phase in range(phases + 1):
            fraction = phase / phases
            row = []
            for k in taps:
                distance = k - fraction
                v = 2 * cutoff * distance
                sinc = 1.0 if abs(v) < 1e-12 else math.sin(math.pi * v) / (math.pi * v)
                window = 0.5 + 0.5 * math.cos(math.pi * distance / half)
                row.append(2 * cutoff * sinc * window)
            table.append(row)
        out_frames = round(n * 8000 / rate)
        for i in range(out_frames):
            position = i * rate / 8000.0
            base = int(position)
            phase = min(phases, round((position - base) * phases))
            weights = table[phase]
            weighted = 0.0
            weight_sum = 0.0
            for offset, coefficient in zip(taps, weights):
                index = base + offset
                if 0 <= index < n:
                    weighted += samples[index] * coefficient
                    weight_sum += coefficient
            sample = weighted / weight_sum if abs(weight_sum) > 1e-12 else 0.0
            output.append(max(-32768, min(32767, round(sample))))

    if sys.byteorder != "little":
        output.byteswap()
    temporary = target.with_name(target.name + ".tmp")
    try:
        with wave.open(str(temporary), "wb") as wav:
            wav.setnchannels(1)
            wav.setsampwidth(2)
            wav.setframerate(8000)
            wav.writeframes(output.tobytes())
        os.chmod(temporary, 0o644)
        os.replace(temporary, target)
    finally:
        temporary.unlink(missing_ok=True)
